1. Add the following policies to `builder.Services.AddAuthorization()`
	```csharp
	builder.Services.AddAuthorization(options =>
	{
	    options.AddPolicy("Astronaut", policy =>
	    {
	        policy.RequireClaim(ClaimTypes.Role, "Pilot");
	        policy.RequireClaim("craft", "Mercury");
	    });
	});
	```

1. Update controllers to use policies
	```csharp
	public class AuthPolicyController : Controller
	{
		[AllowAnonymous]
		public IActionResult Index()
		{
			return View();
		}
		
		[Authorize(Policy = "Astronaut")]
		[HttpGet("AuthPolicy/LaunchSite/CapeCanaveral")]
		public IActionResult CapeCanaveralLaunchSite()
		{
			return Json(new[]
			{
				Craft.Mercury,
			});
		}
	
		[AllowAnonymous]
		[HttpGet("AuthPolicy/LaunchSite/KennedySpaceCentre")]
		public IActionResult KennedySpaceCentreLaunchSite()
		{
			return Json(new[]
			{
				Craft.Apollo,
			});
		}
	}
	```

#### Option 1 - Use Fallback Policy

builder.Services.AddAuthorization(options =>

#### Option 2 - Add Authorization filter
```csharp
builder.Services.AddControllers(config =>
{
    var policy = new AuthorizationPolicyBuilder()
                     .RequireAuthenticatedUser()
                     .Build();
    config.Filters.Add(new AuthorizeFilter(policy));
});
```
